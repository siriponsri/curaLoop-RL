"""Export, fetch and verify every file/link byte without following returned links."""
import argparse,hashlib,json,subprocess,sys,tarfile
from pathlib import Path,PurePosixPath
sys.path.insert(0,str(Path(__file__).resolve().parents[1]))
from tools.sara_control_v1 import ROOT,ACCOUNT,PUBLIC,authority,digest,reserve,settle,write_new
from tools.sara_transport_v1 import source_call

def export():
 r,l=authority('analysis_return');assert not any(b.get('remote') and b['id'] in l['active_jobs'] for b in l['reservations'])
 reserve('remote-owned-export-v1','analysis_return',6000,remote=True,scope='One read-only owned-root export including source/dependencies/intake bytes and symlink metadata; no research')
 result=source_call(ROOT/'tools/sara_return_worker_v1.py',{},ACCOUNT/'export-transport-v1.json',timeout=5700)
 write_new(ACCOUNT/'export-receipt-v1.json',result)
 settle('remote-owned-export-v1','PASS_EXPORTED',measured_export_container_cpu_seconds=result['export_container_cpu_seconds'],measured_remote_host_process_cpu_seconds=result['remote_host_process_cpu_seconds'])
 print(json.dumps(result))

def fetch():
 authority('analysis_return');result=json.loads((ACCOUNT/'export-receipt-v1.json').read_text())
 folder=ACCOUNT/'remote-return-v1';folder.mkdir(exist_ok=False)
 for name in ['owned-return.tar.gz','owned-manifest.json','export-result.json','export.py','container-admission.json','create-receipt.json','container-stdout.log','container-stderr.log']:
  p=subprocess.run(['rtk','proxy','scp','-q','-o','BatchMode=yes','-o','StrictHostKeyChecking=yes','-o','ConnectTimeout=15',
     '100.77.149.83:'+result['export_root']+'/'+name,str(folder/name)],capture_output=True,text=True,timeout=5300)
  write_new(folder/(name+'.transport.json'),dict(exit_code=p.returncode,stderr=p.stderr));assert p.returncode==0
 bundle=folder/'owned-return.tar.gz';mp=folder/'owned-manifest.json'
 assert digest(bundle)==result['bundle_sha256'] and digest(mp)==result['manifest_sha256']
 manifest=json.loads(mp.read_text());seen={};dirs={}
 with tarfile.open(bundle,'r|gz') as tar:
  for member in tar:
   path=PurePosixPath(member.name);assert not path.is_absolute() and '..' not in path.parts
   meta={k:getattr(member,k) for k in ('mode','uid','gid','mtime')}
   if member.isdir():
    assert member.name not in dirs;dirs[member.name]=meta
   elif member.issym():
    assert member.name not in seen;seen[member.name]=dict(type='symlink',target=member.linkname,**meta)
   else:
    assert member.isfile() and member.name not in seen,'Unexpected special/hard-link member'
    with tar.extractfile(member) as f:h=hashlib.file_digest(f,'sha256').hexdigest()
    seen[member.name]=dict(type='file',bytes=member.size,sha256=h,**meta)
 assert seen==manifest['files'] and dirs==manifest['directories']
 value=dict(status='PASS_COMPLETE_OWNED_RETURN_AND_INDEPENDENT_LOCAL_STREAM_REREAD',regular_files=result['regular_files'],
   symlinks=result['symlinks'],directories=result['directories'],regular_bytes=result['regular_bytes'],
   bundle_bytes=bundle.stat().st_size,bundle_sha256=digest(bundle),manifest_sha256=digest(mp),
   every_file_content_and_link_metadata_verified=True,returned_links_extracted_or_followed=False,remote_cleanup='NOT_RUN')
 write_new(PUBLIC/'owned-return-verification-v1.json',value);print(json.dumps(value))

if __name__=='__main__':
 p=argparse.ArgumentParser();p.add_argument('action',choices=['export','fetch']);a=p.parse_args()
 export() if a.action=='export' else fetch()

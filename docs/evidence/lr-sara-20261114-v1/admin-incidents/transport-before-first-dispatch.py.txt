"""Fixed noninteractive SSH transport; uncertain dispatch is never replayed."""
import base64,json,subprocess
from tools.sara_control_v1 import ROOT,ACCOUNT,RELEASE,authority,digest,write_new

def invoke(source,payload,*,timeout=180):
    command='python3 -c "import base64;exec(base64.b64decode(\''+base64.b64encode(source).decode()+'\'))"'
    p=subprocess.run(['rtk','proxy','ssh','-T','-o','BatchMode=yes','-o','StrictHostKeyChecking=yes','-o','ConnectTimeout=15',
                      '100.77.149.83',command],input=json.dumps(payload),capture_output=True,text=True,timeout=timeout)
    return dict(exit_code=p.returncode,stdout=p.stdout,stderr=p.stderr)

def source_call(worker,payload,receipt,*,timeout=180):
    source=Path_read(worker)
    payload={**payload,'release_b64':base64.b64encode(RELEASE.read_bytes()).decode(),'release_sha256':digest(RELEASE),
             'worker_b64':base64.b64encode(source).decode(),'worker_sha256':digest(worker)}
    value=invoke(source,payload,timeout=timeout);write_new(receipt,value)
    assert value['exit_code']==0,'Inspect retained owned state; never replay dispatch blindly'
    return json.loads(value['stdout'])

def Path_read(path):
    return path.read_bytes()

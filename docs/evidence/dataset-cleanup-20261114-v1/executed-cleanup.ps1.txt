function Assert-OwnedPath([string]$path) {

    if (-not $absolute.StartsWith($expectedRoot + '\',[StringComparison]::OrdinalIgnoreCase)) { throw 'Path outside intended repository.' }

    $cursor = Get-Item -LiteralPath $path -Force
    while ($null -ne $cursor) {
        if (($cursor.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0) { throw 'Reparse point denied.' }
        if ([string]::Equals($cursor.FullName,$expectedRoot,[StringComparison]::OrdinalIgnoreCase)) { break }
        $cursor = if ($cursor -is [System.IO.DirectoryInfo]) { $cursor.Parent } else { $cursor.Directory }
    }
    if ($null -eq $cursor) { throw 'Repository ancestor not found.' }
}
